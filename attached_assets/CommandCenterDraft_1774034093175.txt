import React, { useState, useEffect } from 'react';
import { 
  Shield, 
  Search, 
  FileText, 
  Image as ImageIcon, 
  Settings, 
  Zap, 
  ExternalLink, 
  AlertCircle,
  Database,
  PlayCircle,
  TrendingUp,
  Cpu
} from 'lucide-react';

const App = () => {
  const [activeTab, setActiveTab] = useState('dashboard');
  const [researchQuery, setResearchQuery] = useState('');
  const [isResearching, setIsResearching] = useState(false);
  const [researchResults, setResearchResults] = useState(null);
  const [scriptDraft, setScriptDraft] = useState('');

  const BRAND_ORANGE = "#FF5F00";

  // Mock initial data for the dashboard
  const currentTrends = [
    { title: "FCC License Revocation (Iran Coverage)", volume: "High", sentiment: "Volatile" },
    { title: "FACT Act Foreign Disclosures", volume: "Medium", sentiment: "Critical" },
    { title: "Paramount/CNN Merger Funding", volume: "High", sentiment: "Investigative" }
  ];

  const handleResearch = async () => {
    if (!researchQuery) return;
    setIsResearching(true);
    
    // Simulating the Gemini-2.5-flash-preview search/analysis
    const apiKey = ""; // Environment provided
    try {
      const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          contents: [{ parts: [{ text: `Act as an investigative researcher for the channel "Let Me Put You On Game". Find primary source data, FCC filings, or money trails regarding: ${researchQuery}. Provide the "Receipts" in a clear format.` }] }],
          tools: [{ "google_search": {} }]
        })
      });
      const data = await response.json();
      const text = data.candidates?.[0]?.content?.parts?.[0]?.text;
      setResearchResults(text);
    } catch (error) {
      console.error("Research failed", error);
    } finally {
      setIsResearching(false);
    }
  };

  const TabButton = ({ id, icon: Icon, label }) => (
    <button 
      onClick={() => setActiveTab(id)}
      className={`flex flex-col items-center justify-center p-3 transition-all duration-300 border-b-2 ${
        activeTab === id ? 'border-orange-600 text-orange-500 bg-orange-500/10' : 'border-transparent text-gray-400 hover:text-gray-200'
      }`}
    >
      <Icon size={20} />
      <span className="text-[10px] uppercase font-bold mt-1 tracking-tighter">{label}</span>
    </button>
  );

  return (
    <div className="min-h-screen bg-black text-white font-sans overflow-x-hidden pb-20">
      {/* Header */}
      <header className="p-6 border-b border-gray-800 bg-black/80 backdrop-blur-md sticky top-0 z-50 flex justify-between items-center">
        <div>
          <h1 className="text-2xl font-black tracking-tighter flex items-center italic">
            LET ME PUT YOU ON <span className="text-[#FF5F00] ml-2">GAME</span>
          </h1>
          <p className="text-[10px] uppercase text-gray-500 font-bold tracking-[0.2em]">Truth. No Filter. No Apologies.</p>
        </div>
        <div className="relative group">
          <div className="w-12 h-12 rounded-full border-2 border-[#FF5F00] overflow-hidden bg-gray-900 flex items-center justify-center shadow-[0_0_15px_rgba(255,95,0,0.3)]">
             <div className="w-8 h-8 bg-gray-700 rounded-full flex items-center justify-center text-[8px] font-black">LMPYOG</div>
          </div>
          {/* Neon Pulse Overlay */}
          <div className="absolute inset-0 rounded-full border-2 border-[#FF5F00] animate-ping opacity-20 pointer-events-none"></div>
        </div>
      </header>

      {/* Main Content Area */}
      <main className="p-4 max-w-4xl mx-auto space-y-6">
        
        {activeTab === 'dashboard' && (
          <div className="space-y-6 animate-in fade-in slide-in-from-bottom-4">
            {/* Quick Stats */}
            <div className="grid grid-cols-2 gap-4">
              <div className="bg-gray-900/50 p-4 border-l-4 border-[#FF5F00] rounded-r-lg">
                <p className="text-gray-500 text-[10px] uppercase font-bold">Today's Heat</p>
                <h3 className="text-xl font-black">FCC RECOV</h3>
              </div>
              <div className="bg-gray-900/50 p-4 border-l-4 border-[#FF5F00] rounded-r-lg">
                <p className="text-gray-500 text-[10px] uppercase font-bold">Top Scraper</p>
                <h3 className="text-xl font-black">FARA.gov</h3>
              </div>
            </div>

            {/* Live Feed */}
            <section className="bg-gray-900/30 rounded-xl border border-gray-800 p-4">
              <div className="flex items-center gap-2 mb-4">
                <TrendingUp size={18} className="text-[#FF5F00]" />
                <h2 className="text-sm font-black uppercase tracking-widest">Narrative Targets (Live)</h2>
              </div>
              <div className="space-y-3">
                {currentTrends.map((trend, index) => (
                  <div key={index} className="flex justify-between items-center p-3 bg-black/40 border border-gray-800 rounded-lg hover:border-[#FF5F00]/50 transition-colors cursor-pointer group">
                    <span className="text-xs font-medium group-hover:text-[#FF5F00] transition-colors">{trend.title}</span>
                    <span className={`text-[10px] font-bold px-2 py-0.5 rounded ${
                      trend.volume === 'High' ? 'bg-red-500/20 text-red-500' : 'bg-[#FF5F00]/20 text-[#FF5F00]'
                    }`}>
                      {trend.volume}
                    </span>
                  </div>
                ))}
              </div>
            </section>

            {/* Action Card */}
            <div className="bg-gradient-to-br from-[#FF5F00] to-orange-800 p-6 rounded-2xl shadow-xl shadow-orange-900/20">
              <h2 className="text-xl font-black italic mb-2">TIME TO EXPOSE THE GAME.</h2>
              <p className="text-xs text-white/80 mb-4 font-medium leading-relaxed">Start your research on a media parent company or foreign influence node now.</p>
              <button 
                onClick={() => setActiveTab('research')}
                className="w-full bg-black text-white py-3 rounded-lg font-black text-sm uppercase tracking-tighter hover:bg-gray-900 active:scale-95 transition-all"
              >
                Launch Research Terminal
              </button>
            </div>
          </div>
        )}

        {activeTab === 'research' && (
          <div className="space-y-6 animate-in fade-in">
            <div className="relative">
              <input 
                type="text"
                placeholder="Target Media Group / Politician / Agency..."
                className="w-full bg-gray-900 border-2 border-gray-800 rounded-xl p-4 pl-12 focus:border-[#FF5F00] outline-none transition-all font-bold text-sm"
                value={researchQuery}
                onChange={(e) => setResearchQuery(e.target.value)}
              />
              <Search className="absolute left-4 top-1/2 -translate-y-1/2 text-gray-500" size={20} />
            </div>

            <button 
              onClick={handleResearch}
              disabled={isResearching}
              className="w-full bg-[#FF5F00] text-black font-black py-4 rounded-xl flex items-center justify-center gap-2 hover:bg-orange-600 transition-colors disabled:opacity-50"
            >
              {isResearching ? <Cpu className="animate-spin" /> : <Zap size={20} />}
              {isResearching ? "DIGGING FOR RECEIPTS..." : "GET THE GAME"}
            </button>

            {researchResults && (
              <div className="bg-gray-900 border border-[#FF5F00]/30 p-4 rounded-xl font-mono text-xs leading-relaxed overflow-x-auto whitespace-pre-wrap">
                <div className="text-[#FF5F00] font-black uppercase mb-4 flex items-center gap-2 border-b border-gray-800 pb-2">
                  <Database size={14} /> THE RECEIPTS:
                </div>
                {researchResults}
              </div>
            )}
          </div>
        )}

        {activeTab === 'script' && (
          <div className="space-y-6 animate-in fade-in">
             <div className="bg-gray-900 rounded-xl p-4 border border-gray-800">
                <h3 className="text-xs font-black uppercase mb-3 text-gray-400">Episode Script Lab</h3>
                <textarea 
                  className="w-full h-80 bg-black border border-gray-800 rounded-lg p-4 font-mono text-sm outline-none focus:border-[#FF5F00] transition-all resize-none"
                  placeholder="Paste your research here to transform it into a script..."
                  value={scriptDraft}
                  onChange={(e) => setScriptDraft(e.target.value)}
                />
             </div>
             <div className="grid grid-cols-2 gap-4">
                <button className="bg-gray-800 text-white font-black py-3 rounded-lg text-xs uppercase">Save Draft</button>
                <button className="bg-[#FF5F00] text-black font-black py-3 rounded-lg text-xs uppercase">Gen VO Prompt</button>
             </div>
          </div>
        )}

        {activeTab === 'assets' && (
          <div className="grid grid-cols-1 gap-4 animate-in fade-in">
            <div className="bg-gray-900/50 p-6 rounded-2xl border border-gray-800 text-center">
              <div className="w-32 h-32 mx-auto bg-black rounded-full border-2 border-[#FF5F00] flex items-center justify-center mb-4 relative shadow-[0_0_20px_rgba(255,95,0,0.4)]">
                <Shield size={48} className="text-[#FF5F00]" />
                <div className="absolute -bottom-2 -right-2 bg-[#FF5F00] text-black p-1 rounded font-black text-[8px] uppercase">LMPYOG</div>
              </div>
              <h3 className="font-black text-lg">Active Avatar: "Silhouette_V1"</h3>
              <p className="text-xs text-gray-500 mt-1 italic">Tattoos: Orange Neon Glow Enabled</p>
            </div>
            
            <div className="bg-gray-900/30 p-4 rounded-xl border border-gray-800 flex items-center justify-between">
              <span className="text-xs font-bold uppercase tracking-widest text-gray-400">Color Profile</span>
              <div className="flex gap-2">
                <div className="w-8 h-8 rounded bg-[#FF5F00]"></div>
                <div className="w-8 h-8 rounded bg-black border border-gray-700"></div>
                <div className="w-8 h-8 rounded bg-white"></div>
              </div>
            </div>
          </div>
        )}

      </main>

      {/* Bottom Navigation */}
      <nav className="fixed bottom-0 left-0 right-0 bg-black border-t border-gray-800 flex justify-around items-center z-50 h-16 backdrop-blur-lg bg-black/90">
        <TabButton id="dashboard" icon={PlayCircle} label="Intel" />
        <TabButton id="research" icon={Search} label="Receipts" />
        <TabButton id="script" icon={FileText} label="Script" />
        <TabButton id="assets" icon={ImageIcon} label="Brand" />
        <TabButton id="settings" icon={Settings} label="Ops" />
      </nav>
    </div>
  );
};

export default App;
